calculateRR(bool isBuy, float entry, float target, float stop) =>
    if isBuy
        risk = entry - stop
        reward = target - entry
        reward / math.max(0.0001, risk)
    else
        risk = stop - entry
        reward = entry - target
        reward / math.max(0.0001, risk)

// FIXED: Consistent stop price calculation
getStopPrice(bool isBuy, float entry) =>
    atr_value = ta.atr(14)
    
    float stopPrice = na
    if isBuy
        // For buys: ATR-based stop below entry
        stopPrice := entry - (atr_value * 1.5)
    else
        // For sells: ATR-based stop above entry  
        stopPrice := entry + (atr_value * 1.5)
    
    // Ensure stop is valid and not too close
    minDistance = syminfo.mintick * 5  // Minimum 5 pips distance
    if isBuy
        stopPrice := math.min(stopPrice, entry - minDistance)
    else
        stopPrice := math.max(stopPrice, entry + minDistance)
    
    // Round to proper precision
    stopPrice := math.round(stopPrice / syminfo.mintick) * syminfo.mintick
    stopPrice

dirmovement(len) =>
    up = ta.change(high)
    down = -ta.change(low)
    plusDM = na(up) ? na : (up > down and up > 0 ? up : 0)
    minusDM = na(down) ? na : (down > up and down > 0 ? down : 0)
    truerange = ta.tr
    plus = fixnan(100 * ta.rma(plusDM, len) / ta.rma(truerange, len))
    minus = fixnan(100 * ta.rma(minusDM, len) / ta.rma(truerange, len))
    sum = plus + minus
    adx = 100 * ta.rma(math.abs(plus - minus) / (sum == 0 ? 1 : sum), len)
    [adx, plus, minus]

// VIDYA Calculation Function
vidya_calc(src, len, mom, smooth) =>
    momentum = ta.change(src)
    sum_pos = math.sum(momentum >= 0 ? momentum : 0.0, mom)
    sum_neg = math.sum(momentum >= 0 ? 0.0 : -momentum, mom)
    abs_cmo = math.abs(100 * (sum_pos - sum_neg) / (sum_pos + sum_neg))
    alpha = 2 / (len + 1)
    var float vidya_val = src
    vidya_val := alpha * abs_cmo / 100 * src + (1 - alpha * abs_cmo / 100) * nz(vidya_val[1])
    ta.sma(vidya_val, smooth)

    if not useLossProtection

    else
        barsSinceLastLoss = na(lastLossBar) ? 999 : (bar_index - lastLossBar)
        cooldownPassed = barsSinceLastLoss >= lossCooldownBars
        minDistance = useATRBasedDistance ? (atr_value * lossDistanceATRMultiplier) : (minPriceDistanceFromLoss * syminfo.mintick)
        
        recentWinOverride = false
        if useRecentWinOverride and not na(lastWinBar) and not na(lastLossBar) and lastWinDirection == isBuy
            if lastWinBar > lastLossBar
                barsSinceLastWin = bar_index - lastWinBar
                if barsSinceLastWin <= lossCooldownBars
                    recentWinOverride := true
        
        structureChanged = false
        if useMarketStructureCheck and not na(lastLossPrice)
            structureThreshold = lastLossPrice * (structureChangePercent / 100)
            if isBuy
                recentHigh = ta.highest(high, structureChangeBars)
                if high > lastLossPrice + structureThreshold
                    structureChanged := true
            else
                recentLow = ta.lowest(low, structureChangeBars)
                if low < lastLossPrice - structureThreshold
                    structureChanged := true
        
        overrideActive = (structureChanged or recentWinOverride) and barsSinceLastLoss >= (lossCooldownBars * 0.7)
        
        if overrideActive
            reducedCooldown = math.max(10, lossCooldownBars * 0.7)
            reducedDistance = minDistance * 0.8
            
            if barsSinceLastLoss < reducedCooldown
                false
            else
                lossCount = array.size(recentLossEntries)
                tooClose = false
                
                if lossCount > 0
                    for i = 0 to lossCount - 1
                        lossEntry = array.get(recentLossEntries, i)
                        lossBar = array.get(recentLossBars, i)
                        lossDirection = array.get(recentLossDirections, i)
                        
                        if lossDirection == isBuy
                            barsSinceThisLoss = bar_index - lossBar
                            if barsSinceThisLoss < reducedCooldown
                                tooClose := true
                                break
                            
                            priceDistance = math.abs(entryPrice - lossEntry)
                            if priceDistance < reducedDistance
                                tooClose := true
                                break
                
                not tooClose
        else
            if not cooldownPassed
                false
            else
                lossCount = array.size(recentLossEntries)
