calculateRR(bool isBuy, float entry, float target, float stop) =>
    if isBuy
        risk = entry - stop
        reward = target - entry
        reward / math.max(0.01, risk)
    else
        risk = stop - entry
        reward = entry - target
        reward / math.max(0.01, risk)

// Asymmetric stop loss - tighter for sells
getStopPrice(bool isBuy, float entry) =>
    atr_value = ta.atr(atrPeriod)
    swingLow = ta.lowest(low, 10)
    swingHigh = ta.highest(high, 10)
    
    if isBuy
        stopMultiplier = 2.0
        atrStop = entry - atr_value * stopMultiplier
        swingStop = swingLow - atr_value * 0.3
        math.max(swingStop, atrStop)
    else
        stopMultiplier = 1.5  // Tighter stops for sells
        atrStop = entry + atr_value * stopMultiplier
        swingStop = swingHigh + atr_value * 0.2  // Tighter buffer
        math.min(swingStop, atrStop)

vidya_calc(src, len, mom, smooth) =>
    momentum = ta.change(src)
    sum_pos = math.sum(momentum >= 0 ? momentum : 0.0, mom)
    sum_neg = math.sum(momentum >= 0 ? 0.0 : -momentum, mom)
    abs_cmo = math.abs(100 * (sum_pos - sum_neg) / (sum_pos + sum_neg))
    alpha = 2 / (len + 1)
    var float vidya_val = src
    vidya_val := alpha * abs_cmo / 100 * src + (1 - alpha * abs_cmo / 100) * nz(vidya_val[1])
    ta.sma(vidya_val, smooth)

// ──────────────────────── Grouped Win-Rate Tracking (10..120) ────────────────────────
// Keep up to the most recent 120 valid (post-warmup) trade results for dashboard grouping.
// Each group shows winrate for a specific 10-trade block:
//   "Last 10"  = trades 1-10 (most recent)
//   "11-20"    = trades 11-20 (older)
//   "21-30"    = trades 21-30, etc.
// Returns integer 0-100 representing win percentage for that block.
var array<bool> groupedTradeResults = array.new_bool()

pushGroupedResult(bool result) =>
    array.push(groupedTradeResults, result)
    if array.size(groupedTradeResults) > 120
        array.shift(groupedTradeResults)

getGroupedWinRate(int period) =>
    arraySize = array.size(groupedTradeResults)
    groupNumber = period / 10
    startIndex = arraySize - groupNumber * 10
    endIndex = startIndex + 9
    if arraySize >= groupNumber * 10 and startIndex >= 0
        wins = 0
        for i = startIndex to endIndex
            if array.get(groupedTradeResults, i)
                wins += 1
        wins * 10  // Returns 0, 10, 20, ..., 100 (percentage)
    else
        na
