// ──────────────────── Core Functions ────────────────────
calculateRR(bool isBuy, float entry, float target, float stop) =>
    if isBuy
        risk = entry - stop
        reward = target - entry
        reward / math.max(0.0001, risk)
    else
        risk = stop - entry
        reward = entry - target
        reward / math.max(0.0001, risk)

getStopPrice(bool isBuy, float entry) =>
    atr_value = ta.atr(14)
    // Weighted average for "2.5 bars" effect
    swing2Low = ta.lowest(low, 2)
    swing3Low = ta.lowest(low, 3)
    swing2High = ta.highest(high, 2)
    swing3High = ta.highest(high, 3)
    swingLow = (swing2Low + swing3Low) / 2  // Blended 2.5-bar equivalent
    swingHigh = (swing2High + swing3High) / 2  // Blended 2.5-bar equivalent
    
    if isBuy
        atrStop = na(atr_value) ? na : entry - atr_value * 0.8  // Reduced from 1.2 to 0.8 for tighter stops
        keyStop = na(lastKeySupport) ? atrStop : math.max(lastKeySupport, atrStop)
        candidate = na(swingLow) ? keyStop : math.max(swingLow, keyStop)
        nz(candidate, entry - 80 * syminfo.mintick)  // Reduced from 100 to 80
    else
        atrStop = na(atr_value) ? na : entry + atr_value * 0.8  // Reduced from 1.2 to 0.8 for tighter stops
        keyStop = na(lastKeyResistance) ? atrStop : math.min(lastKeyResistance, atrStop)
        candidate = na(swingHigh) ? keyStop : math.min(swingHigh, keyStop)
        nz(candidate, entry + 80 * syminfo.mintick)  // Reduced from 100 to 80

dirmovement(len) =>
    up = ta.change(high)
    down = -ta.change(low)
    plusDM = na(up) ? na : (up > down and up > 0 ? up : 0)
    minusDM = na(down) ? na : (down > up and down > 0 ? down : 0)
    truerange = ta.tr
    plus = fixnan(100 * ta.rma(plusDM, len) / ta.rma(truerange, len))
    minus = fixnan(100 * ta.rma(minusDM, len) / ta.rma(truerange, len))
    sum = plus + minus
    adx = 100 * ta.rma(math.abs(plus - minus) / (sum == 0 ? 1 : sum), len)
    [adx, plus, minus]

[adx, plusDI, minusDI] = dirmovement(14)

// VIDYA Calculation Function
vidya_calc(src, len, mom, smooth) =>
    momentum = ta.change(src)
    sum_pos = math.sum(momentum >= 0 ? momentum : 0.0, mom)
    sum_neg = math.sum(momentum >= 0 ? 0.0 : -momentum, mom)
    abs_cmo = math.abs(100 * (sum_pos - sum_neg) / (sum_pos + sum_neg))
    alpha = 2 / (len + 1)
    var float vidya_val = src
    vidya_val := alpha * abs_cmo / 100 * src + (1 - alpha * abs_cmo / 100) * nz(vidya_val[1])
    ta.sma(vidya_val, smooth)

// Pine Script requires ta.highest/ta.lowest to be called on each bar, so they're calculated above
checkLossProtection(bool isBuy, float entryPrice) =>
    if not useLossProtection
        true  // Protection disabled, allow trade
    else
        // Check cooldown period from last loss
        barsSinceLastLoss = na(lastLossBar) ? 999 : (bar_index - lastLossBar)
        cooldownPassed = barsSinceLastLoss >= lossCooldownBars
        
        // Calculate minimum distance required
        minDistance = useATRBasedDistance ? (atr_value * lossDistanceATRMultiplier) : (minPriceDistanceFromLoss * syminfo.mintick)
        
        // Recent Win Override: Only if win was AFTER the loss and in same direction
        recentWinOverride = false
        if useRecentWinOverride and not na(lastWinBar) and not na(lastLossBar) and lastWinDirection == isBuy
            // Win must be after the loss
            if lastWinBar > lastLossBar
                barsSinceLastWin = bar_index - lastWinBar
                // Win must be recent (within cooldown period) and in same direction
                if barsSinceLastWin <= lossCooldownBars
                    // Only apply override if entry is at least 70% of normal distance (more conservative)
                    recentWinOverride := true
        
        // Market Structure Check: Require significant price movement
        // Use pre-calculated values (calculated on each bar outside function for Pine Script consistency)
        structureChanged = false
        if useMarketStructureCheck and not na(lastLossPrice)
            structureThreshold = lastLossPrice * (structureChangePercent / 100)
            if isBuy
                // For buy: price must have moved significantly above loss level
                if structureChangeHigh > lastLossPrice + structureThreshold
                    structureChanged := true
            else
                // For sell: price must have moved significantly below loss level
                if structureChangeLow < lastLossPrice - structureThreshold
                    structureChanged := true
        
        // Apply overrides only if conditions are met AND still maintain minimum protection
        overrideActive = (structureChanged or recentWinOverride) and barsSinceLastLoss >= (lossCooldownBars * 0.7)  // At least 70% of cooldown must pass
        
        if overrideActive
            // Use more conservative reduced restrictions (80% of normal, not 60%)
            reducedCooldown = math.max(10, lossCooldownBars * 0.7)  // At least 70% of normal cooldown
            reducedDistance = minDistance * 0.8  // 80% of normal distance (more conservative)
            
            if barsSinceLastLoss < reducedCooldown
                false  // Still in reduced cooldown
            else
                // Check with reduced distance
                lossCount = array.size(recentLossEntries)
                tooClose = false
                
                if lossCount > 0
                    for i = 0 to lossCount - 1
                        lossEntry = array.get(recentLossEntries, i)
                        lossBar = array.get(recentLossBars, i)
                        lossDirection = array.get(recentLossDirections, i)
                        
                        if lossDirection == isBuy
                            barsSinceThisLoss = bar_index - lossBar
                            if barsSinceThisLoss < reducedCooldown
                                tooClose := true
                                break
                            
                            priceDistance = math.abs(entryPrice - lossEntry)
                            if priceDistance < reducedDistance
                                tooClose := true
                                break
                
                not tooClose
        else
            // Standard protection logic (no overrides)
            if not cooldownPassed
                false  // Still in cooldown period
            else
                // Check distance from all tracked recent losses (same direction only)
                lossCount = array.size(recentLossEntries)
                tooClose = false
                
                // Only check if there are any losses recorded
                if lossCount > 0
                    for i = 0 to lossCount - 1
                        lossEntry = array.get(recentLossEntries, i)
                        lossBar = array.get(recentLossBars, i)
                        lossDirection = array.get(recentLossDirections, i)
                        
                        // Only check losses in the same direction
                        if lossDirection == isBuy
                            // Check if enough bars have passed since this loss
                            barsSinceThisLoss = bar_index - lossBar
                            if barsSinceThisLoss < lossCooldownBars
                                tooClose := true
                                break
                            
                            // Check price distance
                            priceDistance = math.abs(entryPrice - lossEntry)
                            if priceDistance < minDistance
                                tooClose := true
                                break
                
                not tooClose

            tradeSuccess := false  // Stop hit = always loss
        else
            // For TURTLE or EXPIRED exits: Check if exit price is profitable
            // BUY trade: exitPrice > entry = win, exitPrice < entry = loss
            // SELL trade: exitPrice < entry = win, exitPrice > entry = loss
            if isBuyTrade
                tradeSuccess := exitPrice > currentEntry
            else
                tradeSuccess := exitPrice < currentEntry
        
        if not array.includes(processedTrades, currentTradeId)
            array.push(processedTrades, currentTradeId)
            totalTrades += 1
            
            if totalTrades == 1
                firstTradeTime := time
            
            // Track buy/sell trades for ALL trades (not affected by warmup)
            if isBuyTrade
                totalBuyTrades += 1
                if tradeSuccess
                    buyWins += 1
            else
                totalSellTrades += 1
                if tradeSuccess
                    sellWins += 1
            
            // Win/loss statistics only after warmup period
            if totalTrades > warmupPeriod
                validTradesCount += 1
                array.push(validTradeResults, tradeSuccess)
                
                if tradeSuccess
                    totalWins += 1
                    currentWinStreak += 1
                    currentLossStreak := 0
                    if currentWinStreak > maxWinStreak
                        maxWinStreak := currentWinStreak
                        maxWinStreakTime := time
                else
                    totalLosses += 1
                    currentLossStreak += 1
                    currentWinStreak := 0
                    if currentLossStreak > maxLossStreak
                        maxLossStreak := currentLossStreak
                        maxLossStreakTime := time
                
                // ======================== Loss Protection: Record Losses & Wins ========================
                // Record loss for protection system (only after warmup period)
                if not tradeSuccess and useLossProtection
                    // Add to recent losses tracking
                    array.push(recentLossEntries, currentEntry)
                    array.push(recentLossBars, entryBarIndex)
                    array.push(recentLossDirections, isBuyTrade)
                    
                    // Keep only the most recent losses (limit array size)
                    if array.size(recentLossEntries) > maxRecentLossesToTrack
                        array.shift(recentLossEntries)
                        array.shift(recentLossBars)
                        array.shift(recentLossDirections)
                    
                    // Update last loss tracking
                    lastLossBar := entryBarIndex
                    lastLossPrice := currentEntry
                    lastLossDirection := isBuyTrade
                
                // Record win for override logic (allows trades closer to losses after wins)
                if tradeSuccess and useRecentWinOverride
                    lastWinBar := entryBarIndex
                    lastWinDirection := isBuyTrade
        
        lastTradeCloseBar := bar_index
        lastTradeID := currentTradeId
        
        if show_trade_results
            pointOffset = result_label_offset * syminfo.mintick
            labelText = tradeSuccess ? "WIN\n" + exitReason : "LOSS\n" + exitReason
            label.new(bar_index, 
                     isBuyTrade ? (tradeSuccess ? high + pointOffset : low - pointOffset) : 
                                  (tradeSuccess ? low - pointOffset : high + pointOffset), 
                     labelText, 
                     style=tradeSuccess ? (isBuyTrade ? label.style_label_down : label.style_label_up) : 
                                          (isBuyTrade ? label.style_label_up : label.style_label_down),
                     color=color.new(tradeSuccess ? win_color : loss_color, 0), 
                     textcolor=color.white, 
                     size=result_label_size == 1 ? size.tiny : 
                          result_label_size == 2 ? size.small : 
                          result_label_size == 3 ? size.normal : 
                          result_label_size == 4 ? size.large : size.huge,
                     yloc=yloc.price,
                     textalign=text.align_center)
