// ──────────────────── Core Functions ────────────────────
calculateRR(bool isBuy, float entry, float target, float stop) =>
    risk = isBuy ? entry - stop : stop - entry
    reward = isBuy ? target - entry : entry - target
    reward / math.max(0.0001, risk)

// Turtle-based stop loss (ATR-based) - must be defined before getStopPrice
getTurtleStop(bool isBuy, float entry, float stopATR) =>
    atr_value = ta.atr(atrPeriod)
    atrStop = isBuy ? entry - atr_value * stopATR : entry + atr_value * stopATR
    atrStop

getStopPrice(bool isBuy, float entry, float stopATR) =>
    atr_value = ta.atr(atrPeriod)
    // Use wider swing detection for better stop placement (as per high winrate strategy)
    swingLow = ta.lowest(low, 10)
    swingHigh = ta.highest(high, 10)
    
    // Get Turtle stop as base
    turtleStop = getTurtleStop(isBuy, entry, stopATR)
    
    if isBuy
        atrStop = entry - atr_value * 1.8
        // Prioritize swing low for stop placement (most recent swing low)
        swingStop = na(swingLow) ? atrStop : math.max(swingLow - atr_value * 0.1, atrStop)
        // Use key support if available, otherwise use swing stop
        keyStop = na(lastKeySupport) ? swingStop : math.max(lastKeySupport - atr_value * 0.1, swingStop)
        // Combine with Turtle stop - use the tighter (higher) stop
        combinedStop = math.max(turtleStop, keyStop)
        nz(combinedStop, entry - 120 * syminfo.mintick)
    else
        atrStop = entry + atr_value * 1.8
        // Prioritize swing high for stop placement (most recent swing high)
        swingStop = na(swingHigh) ? atrStop : math.min(swingHigh + atr_value * 0.1, atrStop)
        // Use key resistance if available, otherwise use swing stop
        keyStop = na(lastKeyResistance) ? swingStop : math.min(lastKeyResistance + atr_value * 0.1, swingStop)
        // Combine with Turtle stop - use the tighter (lower) stop
        combinedStop = math.min(turtleStop, keyStop)
        nz(combinedStop, entry + 120 * syminfo.mintick)

// VIDYA Calculation Function
vidya_calc(src, len, mom, smooth) =>
    momentum = ta.change(src)
    sum_pos = math.sum(momentum >= 0 ? momentum : 0.0, mom)
    sum_neg = math.sum(momentum >= 0 ? 0.0 : -momentum, mom)
    sum_total = sum_pos + sum_neg
    abs_cmo = sum_total > 0 ? math.abs(100 * (sum_pos - sum_neg) / sum_total) : 0
    alpha = 2 / (len + 1)
    var float vidya_val = src
    vidya_val := alpha * abs_cmo / 100 * src + (1 - alpha * abs_cmo / 100) * nz(vidya_val[1])
    ta.sma(vidya_val, smooth)

// 勝率計算函數 - 按10筆交易一組統計
getGroupedWinRate(period) =>
    arraySize = array.size(allTradeResults)
    groupNumber = period / 10  // 計算組編號（1-12）
    startIndex = arraySize - groupNumber * 10  // 計算該組的起始索引
    endIndex = startIndex + 9  // 計算該組的結束索引
    
    // 檢查是否有足夠的交易數據
    if arraySize >= groupNumber * 10 and startIndex >= 0
        wins = 0
        // 遍歷該組的10筆交易
        for i = startIndex to endIndex
            if array.get(allTradeResults, i)
                wins += 1
        // 計算百分比，結果必然是10的倍數（0%,10%,...,100%）
        (wins * 10)  // 直接乘以10，避免浮點計算誤差
    else
        na
