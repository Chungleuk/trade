// ──────────────────── Core Functions ────────────────────
calculateRR(bool isBuy, float entry, float target, float stop) =>
    if isBuy
        risk = entry - stop
        reward = target - entry
        reward / math.max(0.0001, risk)
    else
        risk = stop - entry
        reward = entry - target
        reward / math.max(0.0001, risk)

// FIXED: Consistent stop price calculation (Optimized for EURUSD 15M)
getStopPrice(bool isBuy, float entry) =>
    atr_value = ta.atr(14)
    
    float stopPrice = na
    if isBuy
        // For buys: ATR-based stop below entry (EURUSD: Tighter stops, 1.3x ATR)
        stopPrice := entry - (atr_value * 1.3)
    else
        // For sells: ATR-based stop above entry (EURUSD: Tighter stops, 1.3x ATR)
        stopPrice := entry + (atr_value * 1.3)
    
    // Ensure stop is valid and not too close
    minDistance = syminfo.mintick * 5  // Minimum 5 pips distance
    if isBuy
        stopPrice := math.min(stopPrice, entry - minDistance)
    else
        stopPrice := math.max(stopPrice, entry + minDistance)
    
    // Round to proper precision
    stopPrice := math.round(stopPrice / syminfo.mintick) * syminfo.mintick
    stopPrice

// VIDYA Calculation Function
vidya_calc(src, len, mom, smooth) =>
    momentum = ta.change(src)
    sum_pos = math.sum(momentum >= 0 ? momentum : 0.0, mom)
    sum_neg = math.sum(momentum >= 0 ? 0.0 : -momentum, mom)
    abs_cmo = math.abs(100 * (sum_pos - sum_neg) / (sum_pos + sum_neg))
    alpha = 2 / (len + 1)
    var float vidya_val = src
    vidya_val := alpha * abs_cmo / 100 * src + (1 - alpha * abs_cmo / 100) * nz(vidya_val[1])
    ta.sma(vidya_val, smooth)

    maxScore = 0
    
    // Base breakout confirmation (always checked)

    else
        if breakoutBelow
            score := score + 1
    
    // Candlestick patterns (if enabled)
    if useCandlestickPatterns
        maxScore := maxScore + 1
        if isBullish
            if bullishEngulfing or bullishPinBar or bullishHammer or strongBullishCandle
                score := score + 1
        else
            if bearishEngulfing or bearishPinBar or bearishShootingStar or strongBearishCandle
                score := score + 1
    
    // Price structure (if enabled)
    if usePriceStructure
        maxScore := maxScore + 1
        if isBullish
            if bullishStructure
                score := score + 1
        else
            if bearishStructure
                score := score + 1
    
    // Support/Resistance (if enabled)
    if useSupportResistance
        maxScore := maxScore + 1
        if isBullish
            if supportBounce or resistanceBreak
                score := score + 1
        else
            if resistanceBounce or supportBreak
                score := score + 1
    
    // Calculate required score based on strength setting
    float requiredScore = na
    if priceActionStrength == 1
        requiredScore := math.max(1, math.floor(maxScore * 0.4))
    else if priceActionStrength == 2
        requiredScore := math.max(1, math.floor(maxScore * 0.6))
    else
        requiredScore := math.max(1, math.floor(maxScore * 0.8))
    
    score >= requiredScore
