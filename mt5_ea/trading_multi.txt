// ──────────────────── Core Functions ────────────────────
calculateRR(bool isBuy, float entry, float target, float stop) =>
    if isBuy
        risk = entry - stop
        reward = target - entry
        reward / math.max(0.0001, risk)
    else
        risk = stop - entry
        reward = entry - target
        reward / math.max(0.0001, risk)

getStopPrice(bool isBuy, float entry) =>
    float atr_value = ta.atr(atrPeriod)
    swingLow = ta.lowest(low, 5)
    swingHigh = ta.highest(high, 5)
    swingLevel = isBuy ? swingLow : swingHigh
    float atrStop = na
    float keyStop = na
    
    if isBuy
        atrStop := entry - atr_value * 1.7
        keyStop := na(lastKeySupport) ? atrStop : math.max(lastKeySupport, atrStop)
        math.max(swingLevel, keyStop)
    else
        atrStop := entry + atr_value * 1.7
        keyStop := na(lastKeyResistance) ? atrStop : math.min(lastKeyResistance, atrStop)
        math.min(swingLevel, keyStop)

// VIDYA Calculation Function
vidya_calc(float src, int len, int mom, int smooth) =>
    momentum = ta.change(src)
    sum_pos = math.sum(momentum >= 0 ? momentum : 0.0, mom)
    sum_neg = math.sum(momentum >= 0 ? 0.0 : -momentum, mom)
    abs_cmo = math.abs(100 * (sum_pos - sum_neg) / (sum_pos + sum_neg))
    alpha = 2 / (len + 1)
    var float vidya_val = src
    vidya_val := alpha * abs_cmo / 100 * src + (1 - alpha * abs_cmo / 100) * nz(vidya_val[1])
    ta.sma(vidya_val, smooth)

// Turtle Stop Loss Calculation (2x N where N = ATR)
getTurtleStop(bool isBuy, float entry) =>
    float n = atr_value  // N = Average True Range
    if isBuy
        entry - (n * turtleStopMultiplier)  // Stop below entry
    else
        entry + (n * turtleStopMultiplier)  // Stop above entry

// Turtle Target (using RR ratio)
getTurtleTarget(bool isBuy, float entry, float stop) =>
    float risk = math.abs(entry - stop)
    if isBuy
        entry + (risk * rrRatio)
    else
        entry - (risk * rrRatio)
